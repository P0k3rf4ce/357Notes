\begin{question}[Practice 1]
Let $f : \mathbb{R} \to \mathbb{R}$ be a continuous function with compact support. Let $E \in \mathbb{R}$ and $\eta > 0$. Compute,
\[
\lim_{\eta \to 0^+} \mathrm{Im} \left[ \int_{\mathbb{R}} \frac{f(x)}{x - (E + i \eta)} \, dx \right]
\]
If $\eta > 0$ show that the function
\[
E \to \mathrm{Im} \left[ \int_{\mathbb{R}} \frac{f(x)}{x - (E + i \eta)} \, dx \right]
\]
is smooth.
\end{question}
\begin{proof}
Note that the function in the integral is 
\[ \Im\left(\frac{f(x)}{x - (E + i \eta)}\right) = \Im\left(\frac{(x - E + i\eta)f(x)}{(x - E)^2 + \eta^2}\right) = \frac{\eta f(x)}{(x - E)^2 + \eta^2} \]
Since $f$ is continuous with compact support, we have $f$ is only nonzero on $[-N, N]$ and $|f(x)| \leq M$ on this interval. We then get 
\[ \left|\Im\left[ \int_{\mathbb{R}} \frac{f(x)}{x - (E + i \eta)} \, dx \right]\right| \leq \int_{-N}^N \frac{\eta Mdx}{(x - E)^2 + \eta^2} \leq \int_{-N}^N \frac{Mdx}{\eta} = \frac{2NM}{\eta} \to 0 \text{ as } \eta \to 0^+  \]
This proves the first part.

Let $F$ be that second function. Then we have 
\[ F(E) = \int_\R \frac{\eta f(x)}{(x - E)^2 + \eta^2} dx = f * g(E) \]
where $g(t) = \frac{\eta}{t^2 + \eta^2}$. Note that $g$ is smooth. It thus suffices to show that $F' = f * g'$ by repeated application of this. For any $E, h$, we have 
\begin{align*}
    \left|\frac{F(E + h) - F(E)}{h} - f*g'(E)\right|
    &= \left|\frac{1}{h}\int_\R \frac{\eta f(x)}{(x - E - h)^2 + \eta^2} dx - \frac{1}{h}\int_\R \frac{\eta f(x)}{(x - E)^2 + \eta^2} dx - \int_\R g'(E - x)f(x)dx\right| \\
    &\leq 2MN \sup_{x \in [-N, N]}\left|\frac{g(E - x + h) - g(E - x)}{h} - g'(E - x)\right| \to 0 \text{ as } h \to 0
\end{align*}
The latter limit is uniform since $g$ is $C^1$. Hence $F$ is smooth. 
\end{proof}

\begin{question}[Practice 2]
Show that a monotonic function $f : \mathbb{R} \to \mathbb{R}$ is Borel measurable.
\end{question}
\begin{proof}
Without loss of generality, suppose $f$ is increasing. It suffices to show that for any $(a, b) \subseteq \R$, $f^{-1}(a, b)$ is Borel measurable. If it is empty or a singleton then we are done. Otherwise, we show that $f^{-1}(a, b)$ is an interval. For that, we let $x < y \in f^{-1}(a, b)$ and show that any $t \in (x, y)$ is in $f^{-1}(a, b)$. Since $f$ is increasing, we have $f(x) \leq f(t) \leq f(y)$, so $f(t) \in [f(x), f(y)] \subseteq (a, b)$, and hence $t \in f^{-1}(a, b)$. Therefore, $f$ is Borel measurable.
\end{proof}

\begin{question}[Practice 3]
If $m(E) = 0$ what can you say about $L^p(E)$?
\end{question}
\begin{proof}
Note that any set of outer measure zero is measurable, so any function $E \to \cl{\R}$ is measurable. Furthermore, for any such $f$, we have 
\[ \int_E |f|^p = 0 \implies f = 0 \]
This shows that $L^p(E)$ contains only the zero function (under the equivalence relation).
\end{proof}

\begin{question}[Practice 4]
For which $p$ are trigonometric polynomials dense in $L^p(-\pi, \pi)$?
\end{question}
\begin{proof}
Recall that functions in $L^p(-\pi, \pi)$ can be approximated by continuous functions on $(-\pi, \pi)$ of bounded support. If we can show that each continuous function on $[a, b] \subseteq (-\pi, \pi)$ can be approximated by trigonometric polynomials, then we will have our result. One can use a Stone-Weierstrass Theorem argument to show that this can be done with the uniform convergence topology, so we just need to show that uniform convergence on $[a, b]$ implies $L^p$ convergence on $(-\pi, \pi)$, given that the functions are zero a.e. outside $[a, b]$. Suppose conitnuous $f_n: (-\pi, \pi) \to \cl{\R}$ converge to a continuous $f$ uniformly on $[a, b]$ and $f, f_n$ are supported in $[a, b]$. Then, we have 
\[ \int_{(-\pi, \pi)} |f_n - f|^p \leq (b - a)\sup_{x \in [a, b]}|f_n(x) - f(x)| \to 0 \]
as $n \to \infty$, which completes the proof.
\end{proof}
